\begin{frame}{Calibration (1/2): what was changed and why}
\begin{columns}[T]
\begin{column}{0.53\textwidth}
\small
\textbf{Approach:} Python drives Vissim (COM); 1\,857 runs in 3 phases\\[2pt]
Latin hypercube $\rightarrow$ Gaussian-process surrogate per output
$\rightarrow$ Sobol sensitivity $\rightarrow$ constrained optimum $\rightarrow$ validation
with 10 new seeds\\[6pt]
\textbf{Most influential parameters}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item right-turn speed (L4, L8, L1)
  \item desired speed per approach (exits L2, L3, L5, L6)
  \item desired acceleration / deceleration (L7, L1)
  \item bicycle desired speed (L9 to L12)
  \item conflict-area gaps: no influence
\end{itemize}
\textbf{Constraints:} demand served (GEH $\le$ 5, L4 flow), 600~s warm-up added

\begin{frame}{Alternative designs studied (all three)}
\small
Baseline = calibrated model (phase 2), fixed time program 1, constant demand.
20 replications per scenario with \textbf{common random numbers} $\rightarrow$ paired $t$-tests.
\vspace{6pt}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item \textbf{Actuated control}: VisVAP logic, gap 3~s on the detectors, stage 1 (Kungsgatan)
  up to 75~s, stage 2 (Slottsgatan) up to 15~s
  \item \textbf{Incident}: inputs A, C, E, G change every 10 min (Table 1);
  30-min averages equal the baseline, so only the \emph{timing} of demand changes
  \item \textbf{800 bicycles/h} per direction instead of 100
\end{itemize}
\vspace{6pt}
\textbf{Metrics:} average delay and stops per vehicle (level of service), queue length per
approach in 10-min intervals (where and when), speeds and flows at L1 to L12 (link to the
calibration, throughput)
\end{frame}

\begin{frame}{Observations and questions for discussion}
\small
\begin{itemize}\setlength{\itemsep}{5pt}
  \item The calibration target decides the result: speed-only fitting produced a model that was
  ``right for the wrong reason''. Should acceleration data be a standard calibration target?
  \item L1 has a field s.d.\ of 4.7~km/h but L3 only 0.3~km/h: an RMSE treats them equally, the
  $t$-test does not. Which acceptance criterion should a calibration use?
  \item Remaining misfits (L1 deceleration, exit speeds) look structural: turning speeds act only
  after the stop line. When should we change the model instead of its parameters?
  \item The actuated controller gains 26\,\% delay mainly by taking green from the side road.
  Is average delay the right objective, or should side-road waiting be weighted?
  \item Calibrated at 100 bicycles/h, but the bicycle scenario depends on car to cyclist conflicts
  that the data could not calibrate. How far can a calibrated model be extrapolated?
\end{itemize}
\end{frame}
